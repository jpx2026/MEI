# DHCP

## GW - ISC DHCP
echo "127.0.1.1 siai-supergw" | sudo tee -a /etc/hosts
sudo apt update
sudo apt install isc-dhcp-server -y

sudo nano /etc/dhcp/dhcpd.conf

subnet 192.168.0.0 netmask 255.255.255.0 {
         range 192.168.0.100 192.168.0.150; # dynamic range
         default-lease-time 86400;        # 24h lease time
         max-lease-time 90000;            # 25h maximum lease time
         option domain-name-servers 193.136.212.1, 193.136.212.21;
         option domain-name "siai.dei.uc.pt";
         option subnet-mask 255.255.255.0;
         option broadcast-address 192.168.0.255;
         option routers 192.168.0.254;
}

ip addr show
**eth1**
sudo nano /etc/default/isc-dhcp-server
INTERFACESv4="eth1"

sudo dhcpd -t -cf /etc/dhcp/dhcpd.conf
sudo systemctl start isc-dhcp-server
sudo systemctl enable isc-dhcp-server
sudo systemctl status isc-dhcp-server

## GW - Kea DHCP

sudo nano /etc/kea/kea-dhcp4.conf

{
  "Dhcp4": {
    "interfaces-config": {
      "interfaces": [ "eth1" ]
    },
    "lease-database": {
      "type": "memfile",
      "persist": true,
      "name": "/var/lib/kea/kea-leases4.csv",
      "lfc-interval": 3600
    },
    "subnet4": [
      {
        "id": 1,
        "subnet": "192.168.0.0/24",
        "pools": [
          {
            "pool": "192.168.0.100 - 192.168.0.150"
          }
        ],
        "valid-lifetime": 86400,
        "max-valid-lifetime": 90000,
        "option-data": [
          {
            "name": "domain-name-servers",
            "data": "193.136.212.1, 193.136.212.21"
          },
          {
            "name": "domain-name",
            "data": "siai.dei.uc.pt"
          },
          {
            "name": "routers",
            "data": "192.168.0.254"
          }
        ]
      }
    ]
  }
}

sudo systemctl restart kea-dhcp4-server

**Logs do kea**
sudo journalctl -u kea-dhcp4-server -e --no-pager

**Firewall**
sudo nft add table inet filter
sudo nft add chain inet filter input { type filter hook input priority 0 \; policy accept \; }
sudo nft add rule inet filter input udp dport { 67, 68 } accept
sudo nft list ruleset | sudo tee /etc/nftables.conf
sudo systemctl enable --now nftables

**Monitoring**
cat /var/lib/dhcp/dhcpd.leases | grep "^lease" -A 5

**Logs for errors**
sudo journalctl -u isc-dhcp-server --since "10 minutes ago"
